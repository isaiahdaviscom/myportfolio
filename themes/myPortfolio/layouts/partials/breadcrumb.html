{{/* Breadcrumb Partial ────────────────────────────────────────────────────────────────────────────── Renders an accessible, SEO-friendly breadcrumb trail. Skipped on the home page. Usage: {{ partial "breadcrumb.html" . }} */}} {{- if not .IsHome -}} {{- /* Build the crumb slice — Home → (Section) → Page */ -}} {{- $crumbs := slice (dict "title" "Home" "url" "/" "current" false) -}} {{- /* Add section crumb if we're inside a named section */ -}} {{- if and .Section (not .IsSection) -}} {{- $sectionTitle := .Section | humanize | title -}} {{- $sectionURL := printf "/%s/" .Section -}} {{- $crumbs = $crumbs | append (dict "title" $sectionTitle "url" $sectionURL "current" false) -}} {{- end -}} {{- /* Current page crumb — no URL, aria-current="page" */ -}} {{- $crumbs = $crumbs | append (dict "title" (.Params.heading | default .Title) "url" "" "current" true) -}} {{/* ── Structured Data (BreadcrumbList) ── */}} {{/* ── Visible breadcrumb nav ── */}}
    {{- range $i, $c := $crumbs -}}
  1. {{- if $c.current -}} {{ $c.title }} {{- else -}} {{ $c.title }} {{- end -}}
    2. {{- end -}}
{{- end -}}
