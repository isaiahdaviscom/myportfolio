{{/* Breadcrumb — site-level partial (overrides theme). Uses BEM classes styled in src/css/layout/breadcrumb.css. Skipped on the home page. */}} {{- if not .IsHome -}} {{- /* Build crumb slice: Home → (Section) → Current page */ -}} {{- $crumbs := slice (dict "title" "Home" "url" "/" "current" false) -}} {{- /* Add section crumb when inside a section page (not on the section index itself) */ -}} {{- if and .Section (not .IsSection) -}} {{- $sectionTitle := .Section | humanize | title -}} {{- $sectionURL := printf "/%s/" .Section -}} {{- $crumbs = $crumbs | append (dict "title" $sectionTitle "url" $sectionURL "current" false) -}} {{- end -}} {{- /* Current page — no URL, aria-current */ -}} {{- $crumbs = $crumbs | append (dict "title" .Title "url" "" "current" true) -}} {{/* Structured Data (BreadcrumbList) */}}
    {{- range $i, $c := $crumbs -}}
  1. {{- if $c.current -}} {{ $c.title }} {{- else -}} {{ $c.title }} {{- end -}}
    2. {{- end -}}
{{- end -}}
